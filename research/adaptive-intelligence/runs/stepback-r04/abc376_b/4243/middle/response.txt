Step 1: Track the fixed hand position between operations. For each instruction, the moving hand can travel around the ring; choose the shortest path to the target that never requires moving into the fixed hand's part, avoiding the forbidden move into the other hand.
Step 2: Because the ring is small, test both directions (forward and backward) from current hand position to target. For each path, verify that the target side is never blocked by the fixed hand (i.e., do not pass through the fixed hand's part, and do not end at it). Compute the minimal valid step count, and add it to the total.
Step 3: Apply the moves sequentially and sum the minimal operation counts over all instructions, printing the total.

```python
import sys

def main():
    data = sys.stdin.read().strip().split()
    if not data:
        return
    it = iter(data)
    N = int(next(it))
    Q = int(next(it))
    H = []
    T = []
    for _ in range(Q):
        H.append(next(it))
        T.append(int(next(it)))

    l = 1
    r = 2
    total = 0

    def nxt(p, d):
        # d = +1 or -1
        if d == 1:
            return p + 1 if p < N else 1
        else:
            return p - 1 if p > 1 else N

    for i in range(Q):
        h = H[i]
        target = T[i]
        if h == 'L':
            fixed = r
            pos = l
        else:
            fixed = l
            pos = r

        best = None
        for d in (1, -1):
            p = pos
            steps = 0
            valid = True
            while p != target:
                np = nxt(p, d)
                # Destination must not be the fixed hand (other hand not specified)
                if np == fixed:
                    valid = False
                    break
                p = np
                steps += 1
                if steps > N:
                    valid = False
                    break
            if valid and (best is None or steps < best):
                best = steps
        if best is None:
            best = 0
        total += best
        if h == 'L':
            l = target
        else:
            r = target

    print(total)

if __name__ == "__main__":
    main()
