Step 1: Track the fixed hand position between operations. For each instruction, the moving hand can travel around the ring; choose the shortest path to the target that never requires moving into the fixed hand's part, avoiding the forbidden move into the other hand.
Step 2: For a ring of size N, enumerate both possible directions (clockwise and counterclockwise) along the path. For each candidate path, count the number of moves until reaching the target, but forbid stepping onto the fixed hand position. If a candidate is invalid, skip it; otherwise keep the minimum valid cost.
Step 3: Sum the chosen costs over all instructions and print the total.

```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    idx = 0
    N = int(data[idx]); idx += 1
    Q = int(data[idx]); idx += 1

    left = 1
    right = 2

    total = 0

    for _ in range(Q):
        H = data[idx]; T = int(data[idx+1]); idx += 2
        if H == 'L':
            moving = left
            fixed = right
        else:
            moving = right
            fixed = left

        best = None
        for dir in (1, -1):
            pos = moving
            steps = 0
            valid = True
            # Explore paths around the ring until target is reached.
            # Avoid moving into fixed position.
            while True:
                nxt = ((pos - 1 + dir) % N) + 1
                if nxt == fixed:
                    valid = False
                    break
                pos = nxt
                steps += 1
                if pos == T:
                    valid = True
                    break
                if steps > N:
                    valid = False
                    break

            if valid:
                if best is None or steps < best:
                    best = steps

        total += best
        if H == 'L':
            left = T
        else:
            right = T

    print(total)

if __name__ == "__main__":
    main()
