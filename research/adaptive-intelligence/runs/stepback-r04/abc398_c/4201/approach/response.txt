Step 1: Count the frequency of each integer A_i using a dictionary.
Step 2: For each person i, check if their integer has frequency exactly 1; if so, track the maximum integer and its label.
Step 3: If no qualifying person exists, print -1; otherwise print the chosen label.

Final:
```python
import sys

def main():
    data = sys.stdin.read().split()
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))

    freq = {}
    for x in a:
        freq[x] = freq.get(x, 0) + 1

    best_val = None
    best_label = None
    for i, x in enumerate(a, 1):
        if freq[x] == 1:
            if best_val is None or x > best_val:
                best_val = x
                best_label = i

    if best_label is None:
        print(-1)
    else:
        print(best_label)

if __name__ == "__main__":
    main()
