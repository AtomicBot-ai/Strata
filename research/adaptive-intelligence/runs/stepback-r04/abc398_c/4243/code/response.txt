Step 1: Count the frequency of each integer using a hash map.
Step 2: Identify people whose integer occurs exactly once, and among them find the maximum value.
Step 3: Print the label of the person with that maximum integer, or -1 if none exist.

Final:
```python
import sys

def main():
    data = sys.stdin.read().split()
    if not data:
        return
    n = int(data[0])
    a = list(map(int, data[1:1 + n]))
    freq = {}
    for x in a:
        freq[x] = freq.get(x, 0) + 1
    best_val = None
    best_label = -1
    for i in range(n):
        if freq[a[i]] == 1:
            if best_val is None or a[i] > best_val:
                best_val = a[i]
                best_label = i + 1
    print(best_label)

if __name__ == "__main__":
    main()
